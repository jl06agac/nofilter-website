<script>
(function(){
  var head  = document.getElementById('maniHead');
  var slot  = document.getElementById('maniSlot');
  var slotc = document.getElementById('maniSlotc');
  if(!head || !slot || !slotc) return;
  var words = ['workplace','hotel','cafe'];       // inserted adjectives, in order (longest → shortest)
  var HOLD  = 720, MOVE = 460, ROLL = 560, SWAP = 45;
  var GLYPHS = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ';
  var reduce = window.matchMedia && window.matchMedia('(prefers-reduced-motion:reduce)').matches;
  var running = false;

  var meas = document.createElement('span');
  meas.style.cssText = 'position:absolute;left:-9999px;top:0;white-space:pre;visibility:hidden;pointer-events:none;';
  document.body.appendChild(meas);
  function syncMeas(){ var cs = getComputedStyle(head); ['fontFamily','fontWeight','fontStyle','fontSize','letterSpacing','textTransform'].forEach(function(p){ meas.style[p]=cs[p]; }); }
  function charW(ch){ meas.textContent = ch; return meas.getBoundingClientRect().width; }
  function rowWidth(str){ var w=0; for(var i=0;i<str.length;i++) w+=charW(str[i]); return w; }
  function rnd(){ return GLYPHS[(Math.random()*GLYPHS.length)|0]; }
  function wait(ms){ return new Promise(function(r){ setTimeout(r, ms); }); }

  function buildRow(str){
    slotc.innerHTML = '';
    var letters = [];
    for(var i=0;i<str.length;i++){
      var ch = str[i], s = document.createElement('span');
      s.className = 'mani-ch'; s.style.width = charW(ch) + 'px';
      if(ch === ' '){ s.textContent = ' '; }
      else { s.textContent = rnd(); s.dataset.final = ch; s.classList.add('spin'); letters.push(s); }
      slotc.appendChild(s);
    }
    return letters;
  }
  function setWidth(px){ slot.style.width = px + 'px'; return wait(MOVE); }
  function rollIn(letters){
    return new Promise(function(resolve){
      if(letters.length === 0){ resolve(); return; }
      var n = letters.length, start = performance.now(), lastSwap = 0;
      var settleAt = letters.map(function(_, i){ var f = (i + 1) / n; return ROLL * (0.30 + 0.70 * (1 - Math.pow(1 - f, 3))); });
      (function frame(now){
        var t = now - start, doSwap = (now - lastSwap) > SWAP;
        for(var i=0;i<n;i++){ var s = letters[i];
          if(t >= settleAt[i]){ if(s.classList.contains('spin')){ s.textContent = s.dataset.final; s.classList.remove('spin'); } }
          else if(doSwap){ s.textContent = rnd(); } }
        if(doSwap) lastSwap = now;
        if(t < ROLL){ requestAnimationFrame(frame); }
        else { letters.forEach(function(s){ s.textContent = s.dataset.final; s.classList.remove('spin'); }); resolve(); }
      })(performance.now());
    });
  }
  function rest(){ syncMeas(); buildRow(' '); slot.style.width = charW(' ') + 'px'; }
  async function run(){
    if(running) return; running = true; syncMeas();
    var gapW = charW(' ');
    if(reduce){ buildRow(' '); slot.style.width = gapW + 'px'; running = false; return; }
    for(var i=0;i<words.length;i++){
      var str = ' ' + words[i] + ' ';
      if(i === 0){ var L = buildRow(str); await setWidth(rowWidth(str)); await rollIn(L); }
      else { await setWidth(rowWidth(str)); var L2 = buildRow(str); await rollIn(L2); }
      await wait(HOLD);
    }
    await setWidth(gapW); buildRow(' '); running = false;
  }
  rest();
  if('IntersectionObserver' in window){
    var io = new IntersectionObserver(function(entries){ entries.forEach(function(e){ if(e.isIntersecting){ io.disconnect(); if(document.fonts && document.fonts.ready){ document.fonts.ready.then(run); } else { run(); } } }); }, { threshold: 0.6 });
    io.observe(head);
  } else { run(); }
})();
</script>
