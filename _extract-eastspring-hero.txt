<script>
/* ═══ Treacly scroll — lerped wheel damping ═══════════════════════════════════
   Restores the eased, weighted scroll feel. Wheel/trackpad deltas accumulate into
   a target; the page eases toward it each frame. Deliberately inert wherever the
   document is not the thing being scrolled:
     · while the typed intro holds the page (body.tt-lock)
     · in quote-builder focus mode, where #price is itself the scroll container
       (body.nf-focus) — hijacking there would freeze the builder
     · over any nested scrollable element (drawers, the builder's own panes)
     · touch devices and prefers-reduced-motion (left completely alone)
   Native scroll-behavior is switched to auto so CSS smoothing does not ease every
   one of our own frames; explicit scrollIntoView({behavior:'smooth'}) calls still
   animate because the behaviour is named at the call site. */
(function(){
  var root=document.documentElement;
  if(window.matchMedia&&window.matchMedia('(prefers-reduced-motion:reduce)').matches) return;
  if(window.matchMedia&&window.matchMedia('(hover:none)').matches) return;   // touch: native is better
  var EASE=0.085, target=window.scrollY||0, cur=target, running=false;
  root.style.scrollBehavior='auto';
  function maxY(){ return Math.max(0,(document.scrollingElement||root).scrollHeight-window.innerHeight); }
  function locked(){ var b=document.body; return !b || b.classList.contains('tt-lock') || b.classList.contains('nf-focus'); }
  function nestedScrollable(el){
    for(var n=0;el&&el!==document.body&&n<12;el=el.parentElement,n++){
      var cs=getComputedStyle(el), oy=cs.overflowY;
      if((oy==='auto'||oy==='scroll')&&el.scrollHeight>el.clientHeight+2) return true;
    }
    return false;
  }
  function tick(){
    cur+=(target-cur)*EASE;
    if(Math.abs(target-cur)<0.5){ cur=target; window.scrollTo(0,cur); running=false; return; }
    window.scrollTo(0,cur);
    requestAnimationFrame(tick);
  }
  window.addEventListener('wheel',function(e){
    if(locked()||e.ctrlKey||e.metaKey) return;               // ctrl/⌘+wheel is zoom
    if((window.__nfNestedScrollable||nestedScrollable)(e.target)) return;
    e.preventDefault();
    var d=e.deltaY*(e.deltaMode===1?16:(e.deltaMode===2?window.innerHeight:1));
    if(!running) cur=window.scrollY;                          // resync before a new run
    target=Math.max(0,Math.min(maxY(),(running?target:cur)+d));
    if(!running){ running=true; requestAnimationFrame(tick); }
  },{passive:false});
  // keyboard, scrollbar drags and scrollIntoView move the page themselves — adopt their position
  window.addEventListener('scroll',function(){ if(!running){ target=cur=window.scrollY; } },{passive:true});
  window.addEventListener('resize',function(){ if(!running){ target=cur=window.scrollY; } });
})();
</script>

<script>
/* HTL optical centring — anchors the lockup on the INK of the middle word.
   Two corrections are folded into one here. Glyph ink does not sit centred in its advance
   box (at this size "Hold" carries a ~13px left bearing against "line"'s ~6px right), so
   everything is measured with canvas ink metrics (actualBoundingBox*) rather than layout
   boxes. And the target is "the" on the page centre rather than HOLD/LINE balanced about
   it — scheme B. Re-run on resize so it holds at every width. */
(function(){
  var title=document.querySelector('.cover-hero-title.cascade');
  if(!title) return;
  var rows=[].slice.call(title.querySelectorAll('.line-row'));
  if(rows.length!==3) return;
  var ctx=document.createElement('canvas').getContext('2d');
  function bearings(el){
    var cs=getComputedStyle(el);
    ctx.font=cs.fontStyle+' '+cs.fontWeight+' '+cs.fontSize+' '+cs.fontFamily;
    ctx.textAlign='left';
    var m=ctx.measureText((el.textContent||'').trim());
    return {adv:m.width, lb:-m.actualBoundingBoxLeft, rb:m.width-m.actualBoundingBoxRight};
  }
  var applied=0;
  function box(el){ var r=document.createRange(); r.selectNodeContents(el);
    var b=r.getBoundingClientRect(); r.detach&&r.detach(); return b; }
  function step(){
    if(!title.classList.contains('converged')||!title.offsetParent) return 0;
    if(title.classList.contains('htl-travel')) return 0;   // mid-flight: measuring now corrects to a position the words are still leaving
    // Scheme B: the anchor is the MIDDLE word, not the line.
    // Centring the line put "the" half the HOLD/LINE width difference right of centre —
    // and because the three words converge out of a vertical stack where all three were
    // centre-aligned, the eye watches "the" start on the centre line and then drift off
    // it. Anchoring on "the" removes that drift; HOLD and LINE now hang unevenly, which
    // is the trade. Still measured on ink, not advance boxes, so the bearings work holds.
    var mw=rows[1].querySelector('.rise');
    if(!mw) return 0;
    var bm=bearings(mw), bM=box(mw);
    if(!(bM.width>0)||!(bm.adv>0)) return 0;
    var gs=bM.width/bm.adv;                                   // on-screen glyph scale
    var inkC=((bM.left + bm.lb*gs) + (bM.right - bm.rb*gs))/2;
    var corr=window.innerWidth/2 - inkC;
    if(!isFinite(corr)) return 0;
    // ACCUMULATE: inkLeft/inkRight are measured with any previous correction already
    // applied, so the delta must be added to the standing offset — recomputing from
    // scratch each call would converge to half the required shift.
    applied+=corr;
    if(Math.abs(applied)>0.22*window.innerWidth){ applied=0; title.style.transform=''; return 0; }   // sanity guard: the B shift is half the HOLD/LINE width difference, far more than the old 40px bearing trim
    title.style.transform='translateX('+applied.toFixed(2)+'px) translateY(4vh) scale(0.9)';
    return corr;
  }
  // one pass measures with the previous correction baked in, so iterate to the fixed point
  function centre(){ for(var i=0;i<6;i++){ if(Math.abs(step())<0.2) break; } }
  // The converge TRANSITIONS the transform; measuring mid-flight reads an animated
  // position and under-corrects. Always re-baseline from zero once motion has settled.
